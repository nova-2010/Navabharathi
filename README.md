function categoryTransactions(m,c){return txs().filter(t=&gt;monthKey(t.date)===m.key&amp;&amp;t.cat===c)}
 
function categoryActual(m,c){return categoryTransactions(m,c).reduce((a,t)=&gt;a+t.amount,0)}
 
function statusFor(actual,budget){if(actual&gt;budget+0.01)return'over';if(budget&gt;0&amp;&amp;actual&gt;=budget*.85)return'watch';return'ok'}
